# --- Ejecucion ---------------------------------------------------------------

# El framework. Lo importa toda la aplicacion.
# Serie 5.2 LTS porque lo fija el estandar (02-stack.md seccion 1): tiene
# soporte hasta abril de 2028. Hay 6.x publicado; no se sube hasta que el
# estandar lo diga. Sin esto no hay aplicacion. Permanente.
Django==5.2.19

# Driver de PostgreSQL 17, la base propia (config/settings.py, DATABASES).
# El extra [binary] trae la biblioteca ya compilada: evita exigir compilador
# y cabeceras de libpq en cada maquina. Sin esto la aplicacion no arranca.
# Permanente.
psycopg[binary]==3.3.8

# Lectura de .env en config/settings.py. Sin esto no hay configuracion por
# entorno y los secretos terminarian en el codigo, que es justo lo que
# 04-seguridad.md seccion 1 prohibe. Permanente.
django-environ==0.14.0

# Servidor WSGI de produccion: Gunicorn tras Nginx, bajo systemd (deploy/).
# En local no se usa, ahi corre "manage.py runserver".
# El marcador de plataforma no es un capricho: Gunicorn depende de fcntl, que
# solo existe en POSIX. En Windows se instala y despues falla al arrancar.
# Sin esto, la aplicacion no se puede servir en produccion. Permanente.
gunicorn==26.2.0 ; sys_platform != "win32"


# --- Desarrollo --------------------------------------------------------------

# Formateador y linter, en una sola herramienta. Es la compuerta 1 de cada
# fase:  ruff format .  y  ruff check .
# No se importa en tiempo de ejecucion, pero se instala tambien en el servidor
# porque el repositorio es el mismo y la compuerta se puede correr ahi.
# Permanente.
ruff==0.17.4
